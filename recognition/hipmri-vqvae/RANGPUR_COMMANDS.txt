conda activate comp3710lab2

## Confirm software, source data, and script syntax
python -c "import nibabel, torch; print('PyTorch:', torch.__version__); print('nibabel:', nibabel.__version__)"
test -d /home/groups/comp3710/HipMRI_Study_open/keras_slices_data && echo 'HipMRI data is ready'
bash -n slurm/gpu_sanity.sh slurm/train_vae.sh slurm/train_vqvae.sh

## Stage 5: Submit the short GPU sanity job on the verified comp3710 partition
SANITY_JOB=$(sbatch --parsable slurm/gpu_sanity.sh)
SANITY_JOB=${SANITY_JOB%%;*}
echo "$SANITY_JOB"
squeue -j "$SANITY_JOB" -o '%.18i %.12P %.18j %.2t %.10M %R'

## Inspect the completed short job
sacct -j "$SANITY_JOB" --format=JobID,JobName,Partition,State,Elapsed,ExitCode,MaxRSS
tail -n 120 "hipmri_sanity_${SANITY_JOB}.out"
tail -n 120 "hipmri_sanity_${SANITY_JOB}.err"

## Inspect sanity artifacts before choosing the formal epoch and batch budget
cat "outputs/rangpur-sanity-vae-${SANITY_JOB}/summary.json"
cat "outputs/rangpur-sanity-vqvae-${SANITY_JOB}/summary.json"
cat "predictions/rangpur-sanity-vqvae-${SANITY_JOB}/summary.json"

## Stage 6: Submit formal jobs only after the sanity evidence is accepted
VAE_JOB=$(sbatch --parsable slurm/train_vae.sh)
VAE_JOB=${VAE_JOB%%;*}
echo "$VAE_JOB"

## Run VQ-VAE after the baseline succeeds so both jobs do not compete for GPUs
VQVAE_JOB=$(sbatch --parsable --dependency=afterok:"$VAE_JOB" slurm/train_vqvae.sh)
VQVAE_JOB=${VQVAE_JOB%%;*}
echo "$VQVAE_JOB"

## Monitor formal jobs and inspect accounting after completion
squeue -u "$USER" -o '%.18i %.12P %.18j %.2t %.10M %R'
sacct -j "$VAE_JOB,$VQVAE_JOB" --format=JobID,JobName,Partition,State,Elapsed,ExitCode,MaxRSS
tail -n 120 "hipmri_vae_${VAE_JOB}.out"
tail -n 120 "hipmri_vae_${VAE_JOB}.err"
tail -n 120 "hipmri_vqvae_${VQVAE_JOB}.out"
tail -n 120 "hipmri_vqvae_${VQVAE_JOB}.err"

## The scripts automatically resume from last.pt if a requeued job restarts
## Re-run sinfo before future submissions because cluster partitions can change
## Do not evaluate the test split until both final configurations are frozen

## Stage 7: Run exactly once after both validation-selected checkpoints are frozen
TEST_JOB=$(sbatch --parsable slurm/evaluate_test.sh)
TEST_JOB=${TEST_JOB%%;*}
echo "$TEST_JOB"
squeue -j "$TEST_JOB" -o '%.18i %.12P %.18j %.2t %.10M %R'

## Inspect the one-time test job and retain its checkpoint hashes
sacct -j "$TEST_JOB" --format=JobID,JobName,Partition,State,Elapsed,ExitCode,MaxRSS
tail -n 160 "hipmri_test_${TEST_JOB}.out"
tail -n 120 "hipmri_test_${TEST_JOB}.err"
cat predictions/formal-vae-test/summary.json
cat predictions/formal-vqvae-test/summary.json
